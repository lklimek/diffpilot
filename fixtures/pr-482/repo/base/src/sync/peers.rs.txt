//! Peer bookkeeping: liveness checks and scores.

use crate::rpc::client::RpcClient;
use crate::rpc::error::RpcError;
use crate::rpc::transport::Request;

/// One known peer.
pub struct Peer {
    pub address: String,
    pub score: u32,
}

impl Peer {
    /// Marks the peer as answering.
    pub fn mark_alive(&mut self) {
        self.score = self.score.saturating_add(1);
    }

    /// Marks the peer as failed with `reason`.
    pub fn mark_failed(&mut self, reason: impl std::fmt::Display) {
        tracing::info!(peer = %self.address, %reason, "peer failed");
        self.score = 0;
    }
}

impl Prober {
    /// Sends one ping to `peer` and updates its score.
    pub async fn probe(&self, peer: &mut Peer, ping: Request) {
        match self.client.send(ping).await {
            Ok(_) => peer.mark_alive(),
            Err(RpcError::Transport(e)) => peer.mark_failed(e),
            Err(_) => {}
        }
    }
}
