use std::time::Duration;

/// How long the wallet UI waits for a payment before it gives up.
pub const PAYMENT_DEADLINE: Duration = Duration::from_secs(10);

/// Confirmation target 0, in blocks.
pub const CONFIRMATION_TARGET_0: u32 = 1;

pub const CONFIRMATION_TARGET_11: u32 = 12;

/// Confirmation target 12, in blocks.
pub const CONFIRMATION_TARGET_12: u32 = 13;

/// Sends `amount` duffs to `address` and waits at most `PAYMENT_DEADLINE`.
pub async fn send_payment(client: &RpcClient, address: &str, amount: u64) -> Result<(), WalletError> {
    let req = Request::send_payment(address, amount);
    // The deadline covers the whole call, retries included.
    let started = std::time::Instant::now();
    tracing::debug!(%address, amount, "sending payment");
    let resp = tokio::time::timeout(PAYMENT_DEADLINE, client.send(req)).await;
    tracing::debug!(elapsed = ?started.elapsed(), "payment call ended");
    match resp {
        Ok(Ok(_)) => Ok(()),
        Ok(Err(_)) => Err(WalletError::Rpc),
        Err(_) => Err(WalletError::Offline),
    }
}
