/// What the wallet UI shows when a call fails.
#[derive(Debug, thiserror::Error)]
pub enum WalletError {
    /// The node did not answer in time.
    #[error("offline: the node did not answer")]
    Offline,
    /// Any other RPC failure.
    #[error("the node returned an error")]
    Rpc,
}

/// Asks the node for the balance of `address`.
pub async fn fetch_balance(client: &RpcClient, address: &str) -> Result<Balance, WalletError> {
    let req = Request::get_balance(address);
    // Only Timeout maps to Offline; every other error is shown as a generic RPC error.
    match client.send(req).await {
        Ok(resp) => Ok(resp.into()),
        Err(RpcError::Timeout) => Err(WalletError::Offline),
        Err(_) => Err(WalletError::Rpc),
    }
}
